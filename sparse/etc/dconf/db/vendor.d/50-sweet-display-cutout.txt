# sweet has a centred punch-hole front camera. Silica exposes it to the whole UI
# as Screen.topCutout / Screen.hasCutouts, and lipstick-jolla-home already lays
# out around it (Launcher, LauncherFolder, EventsViewList, PulleyMenuBase...),
# but only if this key is set -- otherwise the clock and status bar are drawn
# straight over the camera.
#
# Geometry taken from the LineageOS device overlay rather than measured:
#   device/xiaomi/sweet/overlay/frameworks/base/core/res/res/values/config.xml
#   <string name="config_mainBuiltInDisplayCutout">
#       M -24.5 51.75 a 24.5 24.5 0 1 0 49 0 a 24.5 24.5 0 1 0 -49 0 Z</string>
# That is a circle in dp: radius 24.5, centre x=0 (screen centre), y=51.75.
# ro.sf.lcd_density=440 so dp->px is 440/160 = 2.75, and the panel is 1080x2400:
#   radius   24.5 * 2.75 = 67.4 px
#   centre   x = 1080/2 = 540, y = 51.75 * 2.75 = 142.3 px
#   rect     x = 540-67.4 = 473, y = 142.3-67.4 = 75, w = h = 135
# Format is a list of rectangles, each [x, y, width, height]; libsailfishsilica
# rejects anything else with "Invalid display cutout rectangle".
#
# The group line matters: without it dconf rejects this file with "Key file
# does not start with a group", and then refuses to compile the whole vendor
# db, silently dropping every other file here too (the camera default in
# 50-sweet-camera.txt among them). Silica reads /desktop/sailfish/silica/cutouts.
[desktop/sailfish/silica]
cutouts=[[473,75,135,135]]
